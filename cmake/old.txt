# Incremental resource copying: only copy changed files
file(GLOB_RECURSE RESOURCE_FILES
    "${CMAKE_SOURCE_DIR}/resources/*"
)

foreach(RES IN LISTS RESOURCE_FILES)
    # Relative path inside resources (e.g. textures/player.png)
    file(RELATIVE_PATH REL_PATH "${CMAKE_SOURCE_DIR}/resources" "${RES}")

    # Directory portion (e.g. textures) and filename (e.g. player.png)
    get_filename_component(REL_DIR "${REL_PATH}" DIRECTORY)
    get_filename_component(REL_NAME "${REL_PATH}" NAME)

    # Destination directory and file
    if(REL_DIR STREQUAL "")
        set(DEST_DIR "$<TARGET_FILE_DIR:MyGame>/resources")
    else()
        set(DEST_DIR "$<TARGET_FILE_DIR:MyGame>/resources/${REL_DIR}")
    endif()
    set(DEST_FILE "${DEST_DIR}/${REL_NAME}")

    add_custom_command(
        TARGET MyGame POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "${DEST_DIR}"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${RES}" "${DEST_FILE}"
        COMMENT "Incremental copy: ${REL_PATH}"
    )
endforeach()
